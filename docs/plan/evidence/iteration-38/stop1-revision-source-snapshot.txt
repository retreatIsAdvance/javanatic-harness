/**
 * 每个 session/turn 独占且不可复用的审批租约，宿主须在该轮 finally 中关闭。
 * 关联由构造器显式传入，不推断工具虚线程的上下文；不拥有共享 relay。
 * 恢复面仅为实例内全量快照，revision 不是 R1 seq，也不是 SSE id。
 */
final class RelayApprovalPrompt implements ApprovalPrompt, AutoCloseable {
    private static final long POLL_MILLIS = 50;

    /** 不额外传递原始 arguments；summary 本身可能含参数，不是脱敏边界。 */
    record Pending(String requestId, Id<Session> session, int turn, String toolName, String summary) { }

    /** 不保留历史；完整替换客户端的 pending 列表。 */
    record Snapshot(long revision, List<Pending> pending) {
        Snapshot {
            pending = List.copyOf(pending);
        }
    }

    private record Active(Request request, CompletableFuture<Outcome> future, Pending pending) { }

    // AbortSignal 没有撤销监听 API；回调只持此小句柄的引用盒，ask finally 清空盒。
    private record Cancellation(ApprovalRelay relay, String requestId) { }

    private final Object lock = new Object();
    private final ApprovalRelay relay;
    private final Id<Session> session;
    private final int turn;
    private final Duration timeout;
    private final ConcurrentHashMap<String, Active> active = new ConcurrentHashMap<>();
    private final Set<Subscription> subscriptions = new HashSet<>();
    private long revision;
    private boolean closed;

    RelayApprovalPrompt(ApprovalRelay relay, Id<Session> session, int turn, long configuredSeconds, boolean interactive) {
        this.relay = Objects.requireNonNull(relay);
        this.session = Objects.requireNonNull(session);
        this.turn = turn;
        this.timeout = ApprovalPrompt.effectiveIdleTimeout(configuredSeconds, interactive);
    }

    @Override
    public boolean ask(ApprovalRequest request, AbortSignal signal) {
        signal.checkAbort();
        Active pending = null;
        AtomicReference<Cancellation> cancellation = new AtomicReference<>();
        try {
            Active registered = register(request);
            pending = registered;
            cancellation.set(new Cancellation(relay, registered.request().id()));
            // 登记回调也可能同步执行，必须离开元数据锁；读快照同时排除已完成 future。
            registered.future().thenRun(() -> remove(registered.request().id()));
            signal.onCancel(() -> cancel(cancellation.get()));
            return await(registered, signal);
        } catch (RuntimeException e) {
            signal.checkAbort();
            throw e;
        } finally {
            try {
                cancel(cancellation.getAndSet(null));
            } finally {
                if (pending != null) {
                    remove(pending.request().id());
                }
            }
        }
    }

    private Active register(ApprovalRequest approval) {
        synchronized (lock) {
            checkOpen();
            Request request = relay.open(timeout);
            Active pending = new Active(request, request.outcome().toCompletableFuture(),
                    new Pending(request.id(), session, turn, approval.toolName(), approval.summary()));
            active.put(request.id(), pending);
            revision++;
            publishLocked();
            return pending;
        }
    }

    private boolean await(Active pending, AbortSignal signal) {
        while (true) {
            signal.checkAbort();
            synchronized (lock) {
                checkOpen();
            }
            Optional<Outcome> outcome = relay.poll(pending.request());
            if (outcome.isPresent()) {
                signal.checkAbort();
                synchronized (lock) {
                    checkOpen();
                    return switch (outcome.orElseThrow()) {
                        case ALLOWED -> true;
                        case REJECTED, EXPIRED -> false;
                        case CANCELLED -> throw new AbortedException("approval cancelled");
                    };
                }
            }
            try {
                // 只读 future 用作短等待/取消唤醒；deadline 裁决始终由 relay.poll 负责。
                pending.future().get(POLL_MILLIS, TimeUnit.MILLISECONDS);
            } catch (TimeoutException e) {
                // 默认 onCancel 是 no-op；无论有无监听都继续 checkAbort。
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
                signal.checkAbort();
                throw new AbortedException("approval wait interrupted");
            } catch (ExecutionException e) {
                signal.checkAbort();
                throw new IllegalStateException("approval outcome failure", e);
            }
        }
    }

    private static void cancel(Cancellation cancellation) {
        if (cancellation == null) {
            return;
        }
        try {
            cancellation.relay().cancel(cancellation.requestId());
        } catch (RpcException e) {
            if (e.code() != -32007) {
                throw e;
            }
            // 墓碑已淘汰或共享 relay 已关闭：本租约签发请求的幂等回收，无剩余待撤销项。
        }
    }

    /** 权威读面与 subscribe 共用同一把锁；settle 完成后不再返回旧 pending。 */
    Snapshot snapshot() {
        synchronized (lock) {
            if (pruneCompletedLocked()) {
                publishLocked();
            }
            return snapshotLocked();
        }
    }

    /** 初始快照与登记原子化，无先取快照再订阅的丢失窗口。 */
    Subscription subscribe() {
        synchronized (lock) {
            Snapshot initial = snapshot();
            Subscription subscription = new Subscription(initial);
            if (closed) {
                subscription.finish(initial);
            } else {
                subscriptions.add(subscription);
            }
            return subscription;
        }
    }

    private void remove(String id) {
        synchronized (lock) {
            if (active.remove(id) != null) {
                revision++;
                publishLocked();
            }
        }
    }

    private boolean pruneCompletedLocked() {
        int before = active.size();
        active.values().removeIf(pending -> pending.future().isDone());
        int removed = before - active.size();
        revision += removed;
        return removed != 0;
    }

    private Snapshot snapshotLocked() {
        return new Snapshot(revision, active.values().stream().map(Active::pending).toList());
    }

    private void publishLocked() {
        pruneCompletedLocked();
        Snapshot snapshot = snapshotLocked();
        // 自有容量 1 邮箱，只覆盖引用并唤醒，不运行外部回调、不等待消费者。
        subscriptions.forEach(subscription -> subscription.offer(snapshot));
    }

    private void checkOpen() {
        if (closed) {
            throw new AbortedException("approval lease closed");
        }
    }

    @Override
    public void close() {
        List<Active> pending;
        synchronized (lock) {
            if (closed) {
                return;
            }
            closed = true;
            pending = List.copyOf(active.values());
            revision += active.size();
            active.clear();
            Snapshot last = snapshotLocked();
            subscriptions.forEach(subscription -> subscription.finish(last));
            subscriptions.clear();
        }
        // cancel 会完成 future 并运行观察者；不得带着租约锁进入 relay。
        pending.forEach(item -> cancel(new Cancellation(relay, item.request().id())));
    }

    /** 容量 1 的合并邮箱；结束时送最后空快照并唤醒等待者，撤销后不接收更新。 */
    final class Subscription implements AutoCloseable {
        private Snapshot latest;
        private boolean ended;

        private Subscription(Snapshot initial) {
            latest = initial;
        }

        private synchronized void offer(Snapshot snapshot) {
            if (!ended) {
                latest = snapshot;
                notifyAll();
            }
        }

        private synchronized void finish(Snapshot last) {
            if (!ended) {
                ended = true;
                latest = last;
                notifyAll();
            }
        }

        /** 最后快照只消费一次；结束且已排空时 fail loud，不永久等待。 */
        synchronized Snapshot take() throws InterruptedException {
            while (latest == null && !ended) {
                wait();
            }
            if (latest == null) {
                throw new IllegalStateException("approval subscription closed");
            }
            Snapshot result = latest;
            latest = null;
            return result;
        }

        /** 0 为非阻塞读；到期或结束且排空返回空。 */
        synchronized Optional<Snapshot> poll(Duration timeout) throws InterruptedException {
            if (timeout.isNegative()) {
                throw new IllegalArgumentException("negative subscription timeout");
            }
            long remaining = timeout.toNanos();
            long deadline = System.nanoTime() + remaining;
            while (latest == null && !ended && remaining > 0) {
                TimeUnit.NANOSECONDS.timedWait(this, remaining);
                remaining = deadline - System.nanoTime();
            }
            Snapshot result = latest;
            latest = null;
            return Optional.ofNullable(result);
        }

        @Override
        public void close() {
            synchronized (lock) {
                subscriptions.remove(this);
                finish(new Snapshot(revision, List.of()));
            }
        }
    }
}

package io.javanatic.harness.workbench.rpc;

import com.fasterxml.jackson.databind.node.ObjectNode;

import java.time.Duration;
import java.util.ArrayDeque;
import java.util.List;
import java.util.Objects;
import java.util.Optional;
import java.util.UUID;
import java.util.concurrent.CompletableFuture;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.CompletionStage;
import java.util.concurrent.ExecutionException;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.TimeoutException;
import java.util.function.LongSupplier;
import java.util.function.Supplier;

/**
 * 审批一次性结算：登记时固定单调 deadline，截止瞬间起 fail-closed，不因重连重置。
 * 活跃表只持未决请求；终态墓碑至多 4096 个、保留 10 分钟，淘汰后 -32007。
 * 所有状态迁移同锁线性化，future 一律锁外通知；没有伴生线程。
 * 超时 Duration 由宿主 ApprovalPrompt 单源计算，本类不另设默认等待界。
 */
public final class ApprovalRelay implements AutoCloseable {
    private static final int MAX_RETIRED = 4096;
    private static final long RETENTION_NANOS = Duration.ofMinutes(10).toNanos();

    /** UI 裁决词表；取消归宿主。 */
    public enum Decision { ALLOW, REJECT }

    /** 唯一终态，成功结算后不被迟来的截止或取消覆盖。 */
    public enum Outcome { ALLOWED, REJECTED, EXPIRED, CANCELLED }

    /** relay 签发的不可伪造句柄；观察副本不授予结算权。 */
    public static final class Request {
        private final Object owner;
        private final String id;
        private final Entry entry;

        private Request(Object owner, String id, Entry entry) {
            this.owner = owner;
            this.id = id;
            this.entry = entry;
        }

        public String id() {
            return id;
        }

        public CompletionStage<Outcome> outcome() {
            return entry.future.minimalCompletionStage();
        }
    }

    private static final class Entry {
        private final String id;
        private final CompletableFuture<Outcome> future = new CompletableFuture<>();
        private final boolean bounded;
        private final long deadline;
        private volatile Outcome done;

        Entry(String id, Duration timeout, long now) {
            this.id = id;
            bounded = !timeout.isZero() && !timeout.isNegative();
            deadline = bounded ? now + timeout.toNanos() : 0;
        }
    }

    private record Tombstone(Outcome outcome, long retiredAt) { }

    private final Object owner = new Object();
    private final ConcurrentHashMap<String, Entry> entries = new ConcurrentHashMap<>();
    private final ConcurrentHashMap<String, Tombstone> retired = new ConcurrentHashMap<>();
    // FIFO 索引与两表同归 owner 锁；不保存 Request/future/参数载荷。
    private final ArrayDeque<String> retirementOrder = new ArrayDeque<>();
    private final Supplier<String> ids;
    private final LongSupplier clock;
    private boolean closed;

    public ApprovalRelay() {
        this(() -> UUID.randomUUID().toString(), System::nanoTime);
    }

    ApprovalRelay(Supplier<String> ids) {
        this(ids, System::nanoTime);
    }

    /** 包内测试缝；生产固定 System.nanoTime，不开放全局可变时钟。 */
    ApprovalRelay(Supplier<String> ids, LongSupplier clock) {
        this.ids = Objects.requireNonNull(ids);
        this.clock = Objects.requireNonNull(clock);
    }

    /** @return 登记请求；0/负 Duration 不设限，正值转换溢出则 fail loud。 */
    public Request open(Duration timeout) {
        synchronized (owner) {
            if (closed) {
                throw new IllegalStateException("approval relay closed");
            }
            prune();
            String id = Objects.requireNonNull(ids.get());
            Entry entry = new Entry(id, Objects.requireNonNull(timeout), clock.getAsLong());
            if (retired.containsKey(id) || entries.putIfAbsent(id, entry) != null) {
                throw new IllegalStateException("duplicate approval request id: " + id);
            }
            return new Request(owner, id, entry);
        }
    }

    /** @throws RpcException -32007 未知；-32009 过期/取消；-32008 已结算 */
    public void settle(String requestId, Decision decision) {
        Objects.requireNonNull(decision);
        Entry entry;
        int error;
        synchronized (owner) {
            prune();
            entry = entries.get(requestId);
            if (entry == null) {
                throw missingOrTerminal(requestId);
            }
            expireIfDue(entry);
            error = terminalError(entry.done);
            if (error == 0) {
                finish(entry, decision == Decision.ALLOW ? Outcome.ALLOWED : Outcome.REJECTED);
            }
        }
        notifyOutcome(entry);
        if (error != 0) {
            throw terminalException(requestId, error);
        }
    }

    /** 显式过期；已终结不覆盖。 */
    public void expire(String requestId) {
        terminate(requestId, Outcome.EXPIRED);
    }

    /** 宿主取消；已终结不覆盖。 */
    public void cancel(String requestId) {
        terminate(requestId, Outcome.CANCELLED);
    }

    private void terminate(String id, Outcome outcome) {
        Entry entry;
        synchronized (owner) {
            prune();
            entry = entries.get(id);
            if (entry == null) {
                if (retired.containsKey(id)) {
                    return;
                }
                throw missingOrTerminal(id);
            }
            finish(entry, outcome);
        }
        notifyOutcome(entry);
    }

    /** 调用方必须持有 owner；状态迁移和从活跃表移除是同一原子动作。 */
    private void finish(Entry entry, Outcome outcome) {
        if (entry.done != null) {
            return;
        }
        entry.done = outcome;
        entries.remove(entry.id, entry);
        retired.put(entry.id, new Tombstone(outcome, clock.getAsLong()));
        retirementOrder.addLast(entry.id);
        prune();
    }

    private void expireIfDue(Entry entry) {
        if (entry.done == null && entry.bounded && clock.getAsLong() - entry.deadline >= 0) {
            finish(entry, Outcome.EXPIRED);
        }
    }

    private static void notifyOutcome(Entry entry) {
        Outcome result = entry.done;
        if (result != null) {
            entry.future.complete(result);
        }
    }

    /** 非阻塞检查登记 deadline；未决返回空，完成通知仍在锁外。 */
    public Optional<Outcome> poll(Request request) {
        if (request.owner != owner) {
            throw new IllegalArgumentException("request belongs to another relay");
        }
        Entry entry = request.entry;
        synchronized (owner) {
            expireIfDue(entry);
        }
        notifyOutcome(entry);
        return Optional.ofNullable(entry.done);
    }

    /** 等待登记时的截止；句柄不依赖墓碑，回收后仍可观察原终态。 */
    public Outcome await(Request request) {
        if (request.owner != owner) {
            throw new IllegalArgumentException("request belongs to another relay");
        }
        Entry entry = request.entry;
        while (true) {
            long remaining;
            synchronized (owner) {
                expireIfDue(entry);
                remaining = entry.bounded ? entry.deadline - clock.getAsLong() : Long.MAX_VALUE;
            }
            notifyOutcome(entry);
            try {
                return entry.future.get(Math.max(1, remaining), TimeUnit.NANOSECONDS);
            } catch (TimeoutException e) {
                // 调度只负责唤醒；是否过期仍以登记 deadline 的单调时钟判断。
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
                synchronized (owner) {
                    finish(entry, Outcome.CANCELLED);
                }
                notifyOutcome(entry);
                throw new RpcException(-32603, "interrupted");
            } catch (ExecutionException e) {
                throw new IllegalStateException("approval outcome failure", e);
            }
        }
    }

    /** 活跃表实际大小，不把墓碑计为活跃请求。 */
    public int pendingCount() {
        synchronized (owner) {
            return entries.size();
        }
    }

    private void prune() {
        long now = clock.getAsLong();
        while (!retirementOrder.isEmpty()) {
            String first = retirementOrder.getFirst();
            if (retired.size() <= MAX_RETIRED && now - retired.get(first).retiredAt() < RETENTION_NANOS) {
                break;
            }
            retirementOrder.removeFirst();
            retired.remove(first);
        }
    }

    private RpcException missingOrTerminal(String id) {
        Tombstone tombstone = retired.get(id);
        return tombstone == null
                ? new RpcException(-32007, "approval request not found", data(id))
                : terminalException(id, terminalError(tombstone.outcome()));
    }

    private static int terminalError(Outcome outcome) {
        if (outcome == null) {
            return 0;
        }
        return switch (outcome) {
            case EXPIRED, CANCELLED -> -32009;
            case ALLOWED, REJECTED -> -32008;
        };
    }

    private static RpcException terminalException(String id, int code) {
        return new RpcException(code, code == -32009 ? "approval expired" : "approval already settled", data(id));
    }

    /** 阻止新登记，取消未决并清空表；通知在锁外，close 幂等。 */
    @Override
    public void close() {
        List<Entry> pending;
        synchronized (owner) {
            closed = true;
            pending = List.copyOf(entries.values());
            pending.forEach(entry -> finish(entry, Outcome.CANCELLED));
            retired.clear();
            retirementOrder.clear();
        }
        pending.forEach(ApprovalRelay::notifyOutcome);
    }

    private static ObjectNode data(String requestId) {
        ObjectNode node = RpcJson.node();
        node.put("requestId", requestId);
        return node;
    }
}

package io.javanatic.harness.workbench.rpc;

import com.fasterxml.jackson.core.JsonProcessingException;
import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.fasterxml.jackson.databind.node.ObjectNode;

/**
 * JSON-RPC 2.0 帧层(停 1-S2)：请求解析 → 方法派发 → 结果/错误序列化。
 * 纯函数(String → String)，无状态无 IO，handler 是唯一注入面。
 * 请求型错与协议违例归本层(-32700/-32600/-32602 结构面)；域码与 -32601
 * 由 handler 经 {@link RpcException} 上抛、本层原样翻译；其余
 * RuntimeException 兜底 -32603，堆栈不外泄。v1 请求-响应 only：batch
 * 数组与 id null 通知一律 -32600。
 */
public final class JsonRpcWire {

    /** 方法派发面：method + params → result；域错误抛 {@link RpcException}。 */
    @FunctionalInterface
    public interface Handler {

        /** 执行方法;params 恒非 null(缺省 params 以 NullNode 传入)。 */
        JsonNode call(String method, JsonNode params);
    }

    private static final ObjectMapper MAPPER = new ObjectMapper();

    /** 处理单个 JSON-RPC 2.0 请求体;恒返回合法 JSON 响应(含错误对象)。 */
    public String handle(String body, Handler handler) {
        final JsonNode request;
        try {
            request = MAPPER.readTree(body);
        } catch (JsonProcessingException e) {
            return serialize(new RpcResponse(null, null, new RpcError(-32700, "parse error")));
        }
        if (request == null || !request.isObject()) {
            return serialize(new RpcResponse(null, null, new RpcError(-32600, "invalid request")));
        }
        JsonNode id = request.get("id");
        if (id == null || id.isNull() || !(id.isNumber() || id.isTextual())) {
            return serialize(new RpcResponse(null, null, new RpcError(-32600, "invalid request")));
        }
        JsonNode version = request.get("jsonrpc");
        if (version == null || !version.isTextual() || !"2.0".equals(version.asText())) {
            return serialize(new RpcResponse(id, null, new RpcError(-32600, "invalid request")));
        }
        JsonNode method = request.get("method");
        if (method == null || !method.isTextual()) {
            return serialize(new RpcResponse(id, null, new RpcError(-32600, "invalid request")));
        }
        JsonNode params = request.get("params");
        if (params != null && !params.isObject()) {
            return serialize(new RpcResponse(id, null, new RpcError(-32602, "invalid params")));
        }
        try {
            RpcRequest parsed = new RpcRequest(method.asText(),
                    params == null ? MAPPER.nullNode() : params, id);
            JsonNode result = handler.call(parsed.method(), parsed.params());
            return serialize(new RpcResponse(id, result, null));
        } catch (RpcException e) {
            return serialize(new RpcResponse(id, null, new RpcError(e.code(), e.getMessage(), e.data())));
        } catch (RuntimeException e) {
            return serialize(new RpcResponse(id, null, new RpcError(-32603, "Internal error")));
        }
    }

    private String serialize(RpcResponse response) {
        ObjectNode node = MAPPER.createObjectNode();
        node.put("jsonrpc", "2.0");
        node.set("id", response.id());
        if (response.error() != null) {
            ObjectNode error = node.putObject("error");
            error.put("code", response.error().code());
            error.put("message", response.error().message());
            if (response.error().data() != null) {
                error.set("data", response.error().data());
            }
        } else {
            node.set("result", response.result());
        }
        try {
            return MAPPER.writeValueAsString(node);
        } catch (JsonProcessingException e) {
            throw new IllegalStateException("unreachable: JsonNode serialization", e);
        }
    }
}

import java.util.Locale;

/**
 * it37 停 1:B36/B37 读面信任围栏——纯静态函数,零状态零 IO,包内可见
 * 不导出(module-info 不动)。信任层非 auth 层(B29 单用户假设不变):
 * B29 回环恒绑防网络面,本围栏防浏览器面——SOP 只挡读响应不挡发请求,
 * 恶意网页可从用户浏览器向回环端口发 POST。
 *
 * <p>两层防御正交闭合:① Host/Origin(B36)——Host 白名单精确比对实际
 * 绑定端口,Origin 缺失放行(curl/非浏览器)、存在即须同源;② Content-Type
 * 严格性(B37)——CORS 预检只挡非简单请求,form-encoded/text/plain 等
 * 简单请求形态不触发预检可直达,故 POST 仅收 application/json。
 *
 * <p>全部 fail-closed,拒绝给具名理由。已知边缘:若宿主栈回环解析为
 * ::1,Host {@code [::1]:port} 不在白名单 → 403(白名单按裁决钉死两形,
 * 扩面留重访)。
 */
final class RequestFence {

    private RequestFence() {
    }

    /**
     * Host/Origin 围栏。返回拒绝理由;{@link Optional#empty()} = 放行。
     *
     * @param host   Host 头原值(null = 缺失);lowercase 后精确等于
     *               {@code 127.0.0.1:<port>} 或 {@code localhost:<port>}
     * @param origin Origin 头原值(null = 缺失放行);存在即须精确等于
     *               {@code http://127.0.0.1:<port>} 或 {@code http://localhost:<port>}
     *               (lowercase 比较;scheme 仅 http——本地无 TLS)
     * @param port   实际绑定端口(server.getAddress().getPort())
     */
    static Optional<String> hostOriginRejection(String host, String origin, int port) {
        String normalizedHost = host == null ? "" : host.toLowerCase(Locale.ROOT);
        if (!normalizedHost.equals("127.0.0.1:" + port)
                && !normalizedHost.equals("localhost:" + port)) {
            return Optional.of("host not allowed");
        }
        if (origin != null) {
            String normalizedOrigin = origin.toLowerCase(Locale.ROOT);
            if (!normalizedOrigin.equals("http://127.0.0.1:" + port)
                    && !normalizedOrigin.equals("http://localhost:" + port)) {
                return Optional.of("origin not allowed");
            }
        }
        return Optional.empty();
    }

    /**
     * Content-Type 严格性(B37):媒体类型({@code ;} 前、trim、lowercase)
     * 精确 {@code application/json}——参数(charset 等)放行;缺失/空/
     * form-urlencoded/text/plain/multipart/其余一律拒。
     */
    static boolean jsonContentType(String contentType) {
        if (contentType == null) {
            return false;
        }
        int semi = contentType.indexOf(';');
        String mediaType = (semi < 0 ? contentType : contentType.substring(0, semi))
            .trim()
            .toLowerCase(Locale.ROOT);
        return mediaType.equals("application/json");
    }
}

    private record EventStream(SseEventSource source, SseEncoder encoder) {}

    private void dispatch(HttpExchange exchange) throws IOException {
        if (events.isPresent() && "/events".equals(exchange.getRequestURI().getPath())

            exchange.getRequestHeaders().getFirst("Origin"),

            return;

                return;

                return;

                exchange.close();
                return;

            exchange.getResponseHeaders().set("Cache-Control", "no-store");

            exchange.sendResponseHeaders(405, -1);
            exchange.close();
            return;

            return;

    private void serveEvents(HttpExchange exchange, EventStream stream) throws IOException {
        SseConnection connection;
        try {
            SseRequest request = SseRequest.parse(exchange.getRequestURI().getRawQuery(),
                exchange.getRequestHeaders().getOrDefault("Last-Event-ID", List.of()));
            SseEventSource.Opened opened = Objects.requireNonNull(
                stream.source().open(request.session(), request.afterSeq()), "SSE source returned null");
            try {
                long head = opened.head();
                if (head < -1) {
                    throw new IllegalStateException("invalid SSE source head");
                }
                if (request.afterSeq() > head) {
                    throw new SseEventSource.HttpFailureException(400, "SSE cursor exceeds head");
                }
                connection = new SseConnection(exchange, opened, stream.encoder(), request.afterSeq());
            } catch (RuntimeException e) {
                try {
                    opened.close();
                } catch (RuntimeException closeFailure) {
                    e.addSuppressed(closeFailure);
                }
                throw e;
            }
        } catch (SseEventSource.HttpFailureException e) {
            respondText(exchange, e.status(), e.getMessage());
            return;
        } catch (RuntimeException e) {
            LOG.log(System.Logger.Level.ERROR, "SSE open failed before 200", e);
            respondText(exchange, 500, "SSE source open failed");
            return;
        }
        boolean accepted;
        synchronized (this) {
            accepted = !closed;
            if (accepted) {
                connections.add(connection);
            }
        }
        // open 与 close 的迟到竞态：全局锁外撤销，绝不发出 SSE 200。
        if (!accepted) {
            connection.close();
            return;
        }
        try {
            connection.run();
        } finally {
            connections.remove(connection);
        }
    }

    /** 围栏及 SSE 打开失败响应：具名理由行(text/plain; charset=utf-8)。 */
    private static void respondText(HttpExchange exchange, int status, String reason)
            throws IOException {
        byte[] out = (reason + "\n").getBytes(StandardCharsets.UTF_8);
        exchange.getResponseHeaders().set("Content-Type", "text/plain; charset=utf-8");
        exchange.sendResponseHeaders(status, out.length);
        exchange.getResponseBody().write(out);
        exchange.close();
    }

    /** 幂等终结；撤销所有 SSE 连接并中断 executor，不持全局锁调用外部 close。 */
    @Override
    public void close() {
        List<SseConnection> active;
        HttpServer bound;
        ExecutorService workers;
        synchronized (this) {
            if (closed) {
                return;
            }
            closed = true;
            active = List.copyOf(connections);
            connections.clear();
            bound = server;
            workers = executor;
        }
        RuntimeException failure = null;
        try {
            // exchange.close 会排空未读请求体；先强制终止 socket，不能等客户端补齐正文。
            if (bound != null) {
                bound.stop(0);
            }
        } catch (RuntimeException e) {
            failure = e;
        } finally {
            try {
                for (SseConnection connection : active) {
                    try {
                        connection.close();
                    } catch (RuntimeException e) {
                        if (failure == null) {
                            failure = e;
                        } else {
                            failure.addSuppressed(e);
                        }
                    }
                }
            } finally {
                if (workers != null) {
                    workers.shutdownNow();
                }
            }
        }
        if (failure != null) {
            throw failure;
        }
    }
}

                out.put("kind", "user");
                yield out;

                ObjectNode out = node();
                out.put("kind", "model");
                out.put("provider", model.provider());
                out.put("model", model.model());
                yield out;

            case MessageSource.Tool tool -> {
                ObjectNode out = node();
                out.put("kind", "tool");
                out.put("toolUseId", tool.toolUseId().value());
                yield out;
            }
            case MessageSource.Compaction ignored -> {
                ObjectNode out = node();
                out.put("kind", "compaction");
                yield out;
            }

                "unknown message source: " + source.getClass().getName());
        };
    }

    private static ArrayNode blocks(List<ContentBlock> content) {
        ArrayNode out = MAPPER.createArrayNode();
        for (ContentBlock block : content) {
            out.add(switch (block) {

                    node.put("type", "text");
                    node.put("text", text.text());
                    yield node;
                }
                case ToolUseBlock use -> {
                    ObjectNode node = node();
                    node.put("type", "toolUse");
                    node.put("id", use.id().value());
                    node.put("name", use.name());
                    node.put("arguments", use.arguments());
                    yield node;
                }
                case ToolResultBlock result -> {
                    ObjectNode node = node();
                    node.put("type", "toolResult");
                    node.put("toolUseId", result.toolUseId().value());
                    node.put("content", result.content());
                    node.put("isError", result.isError());
                    yield node;
                }
            });
        }
        return out;
    }

package io.javanatic.harness.workbench.rpc;

import java.time.Duration;
import java.util.ArrayDeque;
import java.util.Deque;
import java.util.concurrent.TimeUnit;

/** 每连接有界编码帧队列；超帽即终结，EOF 则保留已入队帧。 */
final class SseBuffer {
    private static final int MAX_FRAMES = 64;
    private static final int MAX_BYTES = 8 * 1024 * 1024;

    sealed interface Result {}
    record Data(byte[] bytes) implements Result {}
    record Idle() implements Result {}
    record End() implements Result {}

    private enum State { OPEN, EOF, CLOSED }

    private final Deque<byte[]> queue = new ArrayDeque<>();
    private int bytes;
    private State state = State.OPEN;

    synchronized boolean offer(byte[] frame) {
        if (state != State.OPEN) {
            return false;
        }
        if (queue.size() == MAX_FRAMES || frame.length > MAX_BYTES - bytes) {
            close();
            return false;
        }
        queue.addLast(frame);
        bytes += frame.length;
        notifyAll();
        return true;
    }

    synchronized Result poll(Duration timeout) throws InterruptedException {
        long remaining = timeout.toNanos();
        long deadline = System.nanoTime() + remaining;
        while (queue.isEmpty() && state == State.OPEN && remaining > 0) {
            TimeUnit.NANOSECONDS.timedWait(this, remaining);
            remaining = deadline - System.nanoTime();
        }
        if (!queue.isEmpty()) {
            byte[] frame = queue.removeFirst();
            bytes -= frame.length;
            return new Data(frame);
        }
        return state == State.OPEN ? new Idle() : new End();
    }

    synchronized void finish() {
        if (state == State.OPEN) {
            state = State.EOF;
            notifyAll();
        }
    }

    synchronized void close() {
        state = State.CLOSED;
        queue.clear();
        bytes = 0;
        notifyAll();
    }
}

package io.javanatic.harness.workbench.rpc;

import com.sun.net.httpserver.HttpExchange;

import java.io.IOException;
import java.io.OutputStream;
import java.time.Duration;
import java.util.Optional;

/** 一个 source reader 与一个 HTTP writer；socket 不参与生产者队列锁。 */
final class SseConnection implements AutoCloseable {
    private static final System.Logger LOG = System.getLogger(SseConnection.class.getName());
    private static final Duration KEEPALIVE = Duration.ofSeconds(15);

    private final HttpExchange exchange;
    private final SseEventSource.Opened opened;
    private final SseEncoder encoder;
    private final SseBuffer buffer = new SseBuffer();
    private final Thread reader;
    private Thread writer;
    private volatile boolean closed;
    private long lastSeq;

    SseConnection(HttpExchange exchange, SseEventSource.Opened opened, SseEncoder encoder, long afterSeq) {
        this.exchange = exchange;
        this.opened = opened;
        this.encoder = encoder;
        this.lastSeq = afterSeq;
        reader = Thread.ofVirtual().name("sse-reader").unstarted(this::read);
    }

    void run() {
        synchronized (this) {
            if (writer != null) {
                throw new IllegalStateException("SSE connection already started");
            }
            if (closed) {
                return;
            }
            writer = Thread.currentThread();
        }
        try {
            exchange.getResponseHeaders().set("Content-Type", "text/event-stream; charset=utf-8");
            exchange.getResponseHeaders().set("Cache-Control", "no-store");
            exchange.sendResponseHeaders(200, 0);
            exchange.getResponseBody().flush();
            synchronized (this) {
                if (closed) {
                    return;
                }
                reader.start();
            }
            write(exchange.getResponseBody());
        } catch (IOException e) {
            LOG.log(System.Logger.Level.DEBUG, "SSE socket closed", e);
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
        } catch (RuntimeException e) {
            LOG.log(System.Logger.Level.ERROR, "SSE writer failed after 200", e);
        } finally {
            closeAtBoundary();
        }
    }

    private void write(OutputStream output) throws IOException, InterruptedException {
        long heartbeat = System.nanoTime() + KEEPALIVE.toNanos();
        while (!closed) {
            Duration remaining = Duration.ofNanos(Math.max(0, heartbeat - System.nanoTime()));
            switch (buffer.poll(remaining)) {
                case SseBuffer.Data data -> output.write(data.bytes());
                case SseBuffer.Idle ignored -> {
                    // deadline 到期，统一在下面写无 id 的注释，不被业务帧重置。
                }
                case SseBuffer.End ignored -> {
                    return;
                }
            }
            if (System.nanoTime() >= heartbeat) {
                output.write(SseEncoder.keepalive());
                heartbeat = System.nanoTime() + KEEPALIVE.toNanos();
            }
            output.flush();
        }
    }

    private void read() {
        try {
            while (!closed) {
                Optional<SseFrame> next = opened.next();
                if (next.isEmpty()) {
                    buffer.finish();
                    return;
                }
                SseFrame frame = next.orElseThrow();
                if (frame instanceof SseFrame.Persisted persisted) {
                    if (persisted.logged().seq() <= lastSeq) {
                        throw new IllegalStateException("SSE source sequence is not strictly increasing");
                    }
                    lastSeq = persisted.logged().seq();
                }
                if (!buffer.offer(encoder.encode(frame))) {
                    LOG.log(System.Logger.Level.DEBUG, "SSE buffer overflow or connection revoked");
                    closeAtBoundary();
                    return;
                }
            }
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
            closeAtBoundary();
        } catch (RuntimeException e) {
            // 200 已发送，不能再写 HTTP 错误或伪造 R1 帧：具名日志并主动终结连接。
            LOG.log(System.Logger.Level.ERROR, "SSE source/codec failed after 200", e);
            closeAtBoundary();
        }
    }

    private void closeAtBoundary() {
        try {
            close();
        } catch (RuntimeException e) {
            LOG.log(System.Logger.Level.ERROR, "SSE close failed", e);
        }
    }

    @Override
    public void close() {
        Thread activeWriter;
        synchronized (this) {
            if (closed) {
                return;
            }
            closed = true;
            activeWriter = writer;
        }
        buffer.close();
        reader.interrupt();
        if (activeWriter != null && activeWriter != Thread.currentThread()) {
            activeWriter.interrupt();
        }
        // 先唤醒所有等待者，再做外部回调；任何回调失败仍必须关闭 exchange。
        try {
            opened.close();
        } finally {
            if (activeWriter == Thread.currentThread() && !activeWriter.isInterrupted()) {
                closeFromWriter();
            } else {
                exchange.close();
            }
        }
    }

    private void closeFromWriter() {
        // EOF writer 是自有任务：仅请求体排水可中断，不能污染外部 close 调用者或 source 回调。
        Thread.currentThread().interrupt();
        try {
            exchange.getRequestBody().close();
        } catch (IOException e) {
            LOG.log(System.Logger.Level.DEBUG, "SSE unread request body closed", e);
        } finally {
            // 无请求体的正常 EOF 仍需写出末尾 chunk；带着中断写会截断合法响应。
            Thread.interrupted();
            exchange.close();
        }
    }
}

import io.javanatic.harness.session.persistence.SessionCodecRegistry;
import io.javanatic.harness.session.persistence.SessionEventCodec;

import java.nio.charset.StandardCharsets;
import java.util.Objects;

/** codec 原始载荷 → 单行 JSON；用户数据永不插入 SSE 字段行。 */
final class SseEncoder {
    private static final ObjectMapper MAPPER = new ObjectMapper();
    private final SessionCodecRegistry codecs;

    SseEncoder(SessionCodecRegistry codecs) {
        this.codecs = Objects.requireNonNull(codecs, "codecs");
    }

    byte[] encode(SseFrame frame) {
        String wire = switch (frame) {
            case SseFrame.Persisted persisted -> {
                var logged = persisted.logged();
                SessionEvent event = logged.event();
                SessionEventCodec<?> codec = codecs.forType(event.type()).orElseThrow(
                    () -> new IllegalStateException("missing SSE codec: " + event.type()));
                ObjectNode envelope = MAPPER.createObjectNode();
                envelope.put("seq", logged.seq());
                envelope.put("type", event.type());
                envelope.put("ignorable", event.ignorable());
                envelope.set("data", json(write(codec, event)));
                yield "id: " + logged.seq() + "\ndata: " + envelope + "\n\n";
            }
            case SseFrame.Control control -> "data: " + json(control.data()) + "\n\n";
        };
        return wire.getBytes(StandardCharsets.UTF_8);
    }

    static byte[] keepalive() {
        return ": keepalive\n\n".getBytes(StandardCharsets.UTF_8);
    }

    private static <T extends SessionEvent> JsonValue.Obj write(SessionEventCodec<T> codec, SessionEvent event) {
        return codec.write(codec.typeClass().cast(event));
    }

    private static JsonNode json(JsonValue value) {
        return switch (value) {
            case JsonValue.Obj obj -> {
                ObjectNode object = MAPPER.createObjectNode();
                obj.fields().forEach((key, child) -> object.set(key, json(child)));
                yield object;
            }
            case JsonValue.Arr arr -> {
                ArrayNode array = MAPPER.createArrayNode();
                arr.items().forEach(child -> array.add(json(child)));
                yield array;
            }
            case JsonValue.Str str -> MAPPER.getNodeFactory().textNode(str.value());
            case JsonValue.Num num -> MAPPER.getNodeFactory().numberNode(num.value());
            case JsonValue.Bool bool -> MAPPER.getNodeFactory().booleanNode(bool.value());
            case JsonValue.Null ignored -> MAPPER.getNodeFactory().nullNode();
        };
    }
}

import io.javanatic.harness.kernel.brand.Id;
import io.javanatic.harness.session.Session;

import java.net.URLDecoder;
import java.nio.ByteBuffer;
import java.nio.charset.CharacterCodingException;
import java.nio.charset.StandardCharsets;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

/** 仅 SSE 路由使用的严格 query/header 边界；失败统一翻译为具名 400。 */
record SseRequest(Id<Session> session, long afterSeq) {
    static SseRequest parse(String rawQuery, List<String> lastEventIds) {
        Map<String, String> values = new HashMap<>();
        if (rawQuery != null) {
            for (String pair : rawQuery.split("&", -1)) {
                String[] parts = pair.split("=", 2);
                if (parts.length != 2 || values.putIfAbsent(decode(parts[0]), decode(parts[1])) != null) {
                    throw invalid("duplicate or malformed query");
                }
            }
        }
        String session = values.get("session");
        if (session == null || session.isBlank() || session.codePoints().anyMatch(Character::isISOControl)) {
            throw invalid("session");
        }
        if (lastEventIds.size() > 1) {
            throw invalid("multiple Last-Event-ID headers");
        }
        String cursor = lastEventIds.isEmpty() ? values.get("lastSeq") : lastEventIds.getFirst();
        long after = -1;
        if (cursor != null) {
            if (!cursor.matches("-?[0-9]+")) {
                throw invalid("cursor");
            }
            try {
                after = Long.parseLong(cursor);
            } catch (NumberFormatException e) {
                throw invalid("cursor out of range");
            }
            if (after < -1) {
                throw invalid("cursor below -1");
            }
        }
        return new SseRequest(new Id<>(session), after);
    }

    private static String decode(String raw) {
        try {
            // URLDecoder 的 UTF-8 缺省会替换坏字节；先逐字节解码，再用 REPORT 校验 UTF-8。
            String rawBytes = new String(raw.getBytes(StandardCharsets.UTF_8), StandardCharsets.ISO_8859_1);
            String bytes = URLDecoder.decode(rawBytes, StandardCharsets.ISO_8859_1);
            return StandardCharsets.UTF_8.newDecoder().decode(
                ByteBuffer.wrap(bytes.getBytes(StandardCharsets.ISO_8859_1))).toString();
        } catch (IllegalArgumentException | CharacterCodingException e) {
            throw invalid("URL encoding");
        }
    }

    private static SseEventSource.HttpFailureException invalid(String reason) {
        return new SseEventSource.HttpFailureException(400, "invalid SSE " + reason);
    }
}

        }

    private static final class Cursor implements SseEventSource.Opened {
        private final Fixture source;
        private final Thread writer = Thread.currentThread();
        private final long head;
        private final CountDownLatch closed = new CountDownLatch(1);
        private final CountDownLatch waiting = new CountDownLatch(1);
        private final AtomicInteger closeCalls = new AtomicInteger();
        private volatile Thread reader;
        private long index;
        private boolean ended;

        Cursor(Fixture source, long head, long index) {
            this.source = source;
            this.head = head;
            this.index = index;
        }

        @Override
        public long head() {
            return head;
        }

        @Override
        public Optional<SseFrame> next() throws InterruptedException {
            reader = Thread.currentThread();
            synchronized (source) {
                while (!ended && index >= source.log.size() && !source.finished) {
                    waiting.countDown();
                    source.wait();
                }
                return ended || index >= source.log.size() ? Optional.empty() : Optional.of(source.log.get((int) index++));
            }
        }

        @Override
        public void close() {
            closeCalls.incrementAndGet();
            synchronized (source) {
                ended = true;
                source.notifyAll();
            }
            closed.countDown();
        }
    }

        assertThat(buffer.poll(Duration.ZERO)).isInstanceOf(SseBuffer.End.class);
    }

    @Test
    void idleDiffersFromEofAndAbortWakesWaiter() throws Exception {

        AtomicReference<SseBuffer.Result> result = new AtomicReference<>();
        Thread waiter = Thread.ofVirtual().start(() -> {

            buffer.close();

                }
                if (next <= frames) {
                    return Optional.of(data);
                }
                closed.await();
                return Optional.empty();
            }

    private static class Source implements SseEventSource.Opened {
        private final List<SseFrame> frames;
        final AtomicInteger index = new AtomicInteger();
        final CountDownLatch closed = new CountDownLatch(1);
        final AtomicInteger closeCalls = new AtomicInteger();

        Source(List<SseFrame> frames) {
            this.frames = frames;
        }

        @Override
        public long head() {
            return -1;
        }

        @Override
        public Optional<SseFrame> next() throws InterruptedException {
            int next = index.getAndIncrement();
            return next < frames.size() ? Optional.of(frames.get(next)) : Optional.empty();
        }

        @Override
        public void close() {
            closeCalls.incrementAndGet();
            closed.countDown();
        }
    }

    /** 可确定阻塞在 write 的 HTTP 边界，不依赖操作系统发送缓冲大小。 */
    private static final class Exchange extends HttpExchange {
        private final Headers headers = new Headers();
        private final ByteArrayOutputStream output = new ByteArrayOutputStream();
        private final CountDownLatch writeEntered = new CountDownLatch(1);
        private final CountDownLatch released = new CountDownLatch(1);
        private final AtomicInteger closed = new AtomicInteger();
        private final boolean blocked;

        Exchange(boolean blocked) {
            this.blocked = blocked;
        }

        @Override
        public Headers getRequestHeaders() { return headers; }
        @Override
        public Headers getResponseHeaders() { return headers; }
        @Override
        public URI getRequestURI() { return URI.create("/events?session=s1"); }
        @Override
        public String getRequestMethod() { return "GET"; }
        @Override
        public HttpContext getHttpContext() { return null; }
        @Override
        public void close() {
            closed.incrementAndGet();
            released.countDown();
        }
        @Override
        public InputStream getRequestBody() { return InputStream.nullInputStream(); }
        @Override
        public OutputStream getResponseBody() {
            return new OutputStream() {
                @Override
                public void write(int value) throws IOException {
                    writeEntered.countDown();
                    if (blocked) {
                        try {
                            released.await();
                        } catch (InterruptedException e) {
                            Thread.currentThread().interrupt();
                            throw new IOException("writer interrupted", e);
                        }
                        throw new IOException("exchange closed");
                    }
                    output.write(value);
                }
            };
        }
        @Override
        public void sendResponseHeaders(int code, long length) { }
        @Override
        public InetSocketAddress getRemoteAddress() { return new InetSocketAddress(0); }
        @Override
        public int getResponseCode() { return 200; }
        @Override
        public InetSocketAddress getLocalAddress() { return new InetSocketAddress(0); }
        @Override

        @Override

        @Override
        public void setAttribute(String name, Object value) { }
        @Override
        public void setStreams(InputStream input, OutputStream output) { }
        @Override

    }
